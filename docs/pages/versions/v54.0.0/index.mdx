 import React, { useState } from "react";
import {
  View,
  Text,
  TextInput,
  TouchableOpacity,
  FlatList,
  StyleSheet,
} from "react-native";
import * as FileSystem from "expo-file-system";
import * as Sharing from "expo-sharing";
import * as XLSX from "xlsx";
import * as Print from "expo-print";

export default function App() {
  const [products, setProducts] = useState([]);
  const [name, setName] = useState("");
  const [buy, setBuy] = useState("");
  const [sell, setSell] = useState("");
  const [qty, setQty] = useState("");

  const addProduct = () => {
    if (!name || !buy || !sell || !qty) {
      alert("Hamma maydonni to'ldir!");
      return;
    }
    const item = {
      id: Date.now().toString(),
      name,
      buy: parseFloat(buy),
      sell: parseFloat(sell),
      qty: parseInt(qty),
    };
    setProducts([...products, item]);
    setName("");
    setBuy("");
    setSell("");
    setQty("");
  };

  const deleteProduct = (id) => {
    setProducts(products.filter((p) => p.id !== id));
  };

  const calcProfit = (p) => (p.sell - p.buy) * p.qty;

  const totalProfit = products.reduce((sum, p) => sum + calcProfit(p), 0);

  // 📂 Excelga eksport
  const exportToExcel = async () => {
    if (products.length === 0) {
      alert("Avval mahsulot qo‘shing!");
      return;
    }

    const data = products.map((p) => ({
      Mahsulot: p.name,
      Olish_Narxi: p.buy,
      Sotish_Narxi: p.sell,
      Miqdor: p.qty,
      Foyda: calcProfit(p),
    }));

    data.push({
      Mahsulot: "Umumiy",
      Olish_Narxi: "",
      Sotish_Narxi: "",
      Miqdor: "",
      Foyda: totalProfit,
    });

    const ws = XLSX.utils.json_to_sheet(data);
    const wb = XLSX.utils.book_new();
    XLSX.utils.book_append_sheet(wb, ws, "Savdo");

    const wbout = XLSX.write(wb, { type: "base64", bookType: "xlsx" });
    const fileUri = FileSystem.documentDirectory + "savdo.xlsx";
    await FileSystem.writeAsStringAsync(fileUri, wbout, {
      encoding: FileSystem.EncodingType.Base64,
    });

    await Sharing.shareAsync(fileUri, {
      mimeType:
        "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
      dialogTitle: "Savdo hisoboti (Excel)",
      UTI: "com.microsoft.excel.xlsx",
    });
  };

  // 📑 PDFga eksport
  const exportToPDF = async () => {
    if (products.length === 0) {
      alert("Avval mahsulot qo‘shing!");
      return;
    }

    let rows = products
      .map(
        (p) =>
          `<tr>
            <td>${p.name}</td>
            <td>${p.buy}</td>
            <td>${p.sell}</td>
            <td>${p.qty}</td>
            <td>${calcProfit(p)}</td>
          </tr>`
      )
      .join("");

    rows += `<tr style="font-weight:bold; background:#f2f2f2">
        <td>Umumiy</td>
        <td></td>
        <td></td>
        <td></td>
        <td>${totalProfit}</td>
      </tr>`;

    const html = `
      <html>
        <head>
          <meta charset="utf-8" />
          <style>
            body { font-family: Arial, sans-serif; padding:20px; }
            h1 { text-align:center; color:#333; }
            table { width:100%; border-collapse: collapse; margin-top:20px; }
            th, td { border:1px solid #ccc; padding:8px; text-align:center; }
            th { background:#4CAF50; color:white; }
          </style>
        </head>
        <body>
          <h1>🛒 Savdo Hisoboti</h1>
          <table>
            <tr>
              <th>Mahsulot</th>
              <th>Olish narxi</th>
              <th>Sotish narxi</th>
              <th>Miqdor</th>
              <th>Foyda</th>
            </tr>
            ${rows}
          </table>
        </body>
      </html>
    `;

    const { uri } = await Print.printToFileAsync({ html });

    await Sharing.shareAsync(uri, {
      mimeType: "application/pdf",
      dialogTitle: "Savdo hisoboti (PDF)",
      UTI: "com.adobe.pdf",
    });
  };

  return (
    <View style={styles.container}>
      <Text style={styles.h}>🛒 Bozor Savdo App</Text>

      <TextInput
        style={styles.input}
        placeholder="Mahsulot nomi"
        value={name}
        onChangeText={setName}
      />
      <TextInput
        style={styles.input}
        placeholder="Olish narxi"
        value={buy}
        onChangeText={setBuy}
        keyboardType="numeric"
        inputMode="numeric"
      />
      <TextInput
        style={styles.input}
        placeholder="Sotish narxi"
        value={sell}
        onChangeText={setSell}
        keyboardType="numeric"
        inputMode="numeric"
      />
      <TextInput
        style={styles.input}
        placeholder="Miqdor"
        value={qty}
        onChangeText={setQty}
        keyboardType="numeric"
        inputMode="numeric"
      />

      <TouchableOpacity style={styles.btn} onPress={addProduct}>
        <Text style={styles.btnText}>➕ Qo'shish</Text>
      </TouchableOpacity>

      <FlatList
        style={{ marginTop: 15 }}
        data={products}
        keyExtractor={(item) => item.id}
        renderItem={({ item }) => (
          <View style={styles.card}>
            <Text style={styles.cardTitle}>
              {item.name} | {item.qty} dona
            </Text>
            <Text>Olish: {item.buy} so'm</Text>
            <Text>Sotish: {item.sell} so'm</Text>
            <Text style={styles.profit}>
              💰 Foyda: {calcProfit(item).toLocaleString()} so'm
            </Text>
            <TouchableOpacity onPress={() => deleteProduct(item.id)}>
              <Text style={styles.delete}>❌ O‘chirish</Text>
            </TouchableOpacity>
          </View>
        )}
      />

      {products.length > 0 && (
        <Text style={styles.total}>
          🔥 Umumiy foyda: {totalProfit.toLocaleString()} so'm
        </Text>
      )}

      {products.length > 0 && (
        <View style={{ marginTop: 15 }}>
          <TouchableOpacity
            style={[styles.btn, { backgroundColor: "#1976D2", marginBottom: 10 }]}
            onPress={exportToExcel}
          >
            <Text style={styles.btnText}>📂 Excelga eksport</Text>
          </TouchableOpacity>

          <TouchableOpacity
            style={[styles.btn, { backgroundColor: "#E64A19" }]}
            onPress={exportToPDF}
          >
            <Text style={styles.btnText}>📑 PDFga eksport</Text>
          </TouchableOpacity>
        </View>
      )}
    </View>
  );
}

const styles = StyleSheet.create({
  container: {
    flex: 1,
    padding: 20,
    marginTop: 40,
    backgroundColor: "#f4f6f9",
  },
  h: {
    fontSize: 24,
    fontWeight: "bold",
    marginBottom: 15,
    textAlign: "center",
    color: "#333",
  },
  input: {
    borderWidth: 1,
    borderColor: "#ccc",
    marginBottom: 10,
    padding: 10,
    borderRadius: 8,
    backgroundColor: "#fff",
  },
  btn: {
    padding: 12,
    borderRadius: 8,
    alignItems: "center",
  },
  btnText: {
    color: "#fff",
    fontWeight: "bold",
    fontSize: 16,
  },
  card: {
    backgroundColor: "#fff",
    padding: 15,
    marginBottom: 10,
    borderRadius: 10,
    shadowColor: "#000",
    shadowOpacity: 0.1,
    shadowOffset: { width: 0, height: 2 },
    shadowRadius: 5,
    elevation: 3,
  },
  cardTitle: {
    fontSize: 18,
    fontWeight: "600",
    marginBottom: 5,
    color: "#222",
  },
  profit: {
    marginTop: 5,
    fontWeight: "bold",
    color: "#2E7D32",
  },
  delete: {
    marginTop: 8,
    color: "red",
    fontWeight: "bold",
  },
  total: {
    fontSize: 18,
    fontWeight: "bold",
    textAlign: "center",
    marginTop: 20,
    color: "#d32f2f",
  },
});
